Welcome to Kyrazo! This guide will help you get your first event dispatched from your system to a webhook target.

## 1. Get your API Key

To authenticate your requests, you'll need an API Key.

1.  Log in to the [Kyrazo Dashboard](https://kyrazo.com/dashboard).
2.  Navigate to the **Secrets Page** in the sidebar.
3.  Click **Generate New Key**.
4.  Give it a name and select the necessary **Scopes** (e.g., `Dispatch Event`).
5.  Copy and securely store your **API Key**. You won't be able to see it again!

<Note>
  Keep your API Key secure. If it's ever compromised, you can rotate it from the settings page.
</Note>

## 2. Create a Target

A Target is where Kyrazo will deliver your events.

1.  In the dashboard, go to the **Targets** section.
2.  Click **Create Target**.
3.  Enter a name (e.g., "My API") and a destination URL. If you don't have one yet, you can use a tool like [Webhook.site](https://webhook.site) to test.
4.  Save the target and note down its **Target ID**.

## 3. Dispatch your first event

Now you're ready to send data. Use one of our official SDKs or a simple `curl` command to dispatch your first event.

<Tabs>
  <Tab title="Node.js">
    ```typescript
    import { Kyrazo } from "@kyrazo/sdk";

    const kyrazo = new Kyrazo({ apiKey: "{YOUR_API_KEY}" });

    await kyrazo.events.single("{NAMESPACE_ID}", {
      eventType: "user.signup",
      payload: { userId: "123", email: "user@example.com" },
      targets: [{ targetId: "{YOUR_TARGET_ID}" }]
    });
    ```
  </Tab>
  <Tab title="Python">
    ```python
    from kyrazo import Kyrazo

    client = Kyrazo(api_key="{YOUR_API_KEY}")

    client.events.publish(
        namespace_id="{NAMESPACE_ID}",
        body={
            "eventType": "user.signup",
            "payload": {"userId": "123", "email": "user@example.com"},
            "targets": [{"targetId": "{YOUR_TARGET_ID}"}]
        }
    )
    ```
  </Tab>
  <Tab title="Go">
    ```go
    package main

    import (
    	"context"
    	"fmt"
    	"log"

    	"github.com/kyrazolabs/sdks/go"
    	"github.com/kyrazolabs/sdks/go/models"
    )

    func main() {
    	client := kyrazo.New("{YOUR_API_KEY}")

    	resp, err := client.Events.Single(context.Background(), "{NAMESPACE_ID}", models.PublishEvent{
    		EventType: "user.signup",
    		Payload: map[string]interface{}{
    			"userId": "123",
    			"email":  "user@example.com",
    		},
    		Targets: []models.EventTarget{
    			{TargetId: "{YOUR_TARGET_ID}"},
    		},
    	})

    	if err != nil {
    		log.Fatalf("failed to dispatch: %v", err)
    	}
    	fmt.Printf("Dispatched: %s\n", resp.EventId)
    }
    ```
  </Tab>
  <Tab title="cURL">
    ```bash

         -H "x-api-key: {YOUR_API_KEY}" \
         -H "Content-Type: application/json" \
         -d '{
           "eventType": "user.signup",
           "data": {
             "userId": "123",
             "email": "user@example.com"
           },
           "targets": [
             { "targetId": "{YOUR_TARGET_ID}" }
           ]
         }'
    ```
  </Tab>
</Tabs>

## 4. Receive and verify webhooks

When Kyrazo delivers an event to your target, you should verify its signature to ensure it's authentic.

<Tabs>
  <Tab title="Node.js">
    ```javascript
    const { Webhook } = require("@kyrazo/sdk");
    const express = require("express");
    const app = express();

    // Kyrazo sends the raw request body for signature verification
    app.post("/webhook", express.text({ type: "application/json" }), (req, res) => {
      try {
        Webhook.verify(
          req.body, 
          req.headers, 
          "YOUR_TARGET_SECRET"
        );
        console.log("Verified webhook:", JSON.parse(req.body));
        res.status(200).send("OK");
      } catch (err) {
        res.status(401).send("Unauthorized");
      }
    });
    ```
  </Tab>
  <Tab title="Python (Flask)">
    ```python
    from flask import Flask, request, abort
    from kyrazo import Webhook, ValidationError

    app = Flask(__name__)

    @app.route("/webhook", methods=["POST"])
    def webhook():
        payload = request.get_data(as_text=True)
        headers = dict(request.headers)
        
        try:
            Webhook.verify(payload, headers, "YOUR_TARGET_SECRET")
            return "OK", 200
        except ValidationError:
            abort(401)
    ```
  </Tab>
  <Tab title="Go">
    ```go
    package main

    import (
    	"crypto/hmac"
    	"crypto/sha256"
    	"encoding/hex"
    	"io"
    	"net/http"
    )

    func main() {
    	http.HandleFunc("/webhook", func(w http.ResponseWriter, r *http.Request) {
    		body, _ := io.ReadAll(r.Body)
    		signature := r.Header.Get("X-Kyrazo-Signature")
    		secret := []byte("YOUR_TARGET_SECRET")

    		mac := hmac.New(sha256.New, secret)
    		mac.Write(body)
    		expected := hex.EncodeToString(mac.Sum(nil))

    		if hmac.Equal([]byte(signature), []byte(expected)) {
    			w.WriteHeader(http.StatusOK)
    		} else {
    			http.Error(w, "Unauthorized", http.StatusUnauthorized)
    		}
    	})
    	http.ListenAndServe(":8080", nil)
    }
    ```
  </Tab>
</Tabs>

## 5. Verify Delivery

Once you send the request, you can verify the result:

1.  Check your destination URL (or Webhook.site) to see the incoming payload.
2.  In the Kyrazo Dashboard, navigate to the **Dispatch/Ingest Logs** for your namespace.
3.  You should see your event listed with a status of `Queued` or `Delivered`, along with detailed logs for the delivery attempt.
